// 발광 눈: 거리 함수(SDF)로 모양을 계산해 표정끼리 섞음
float r    = mix(.026, .045, uSurp) * mix(1., .7, uScared);
float base = fill(length(q) - r);               // 점
float arc  = fill(abs(length(h) - .03) - .0075)
           * step(.006, h.y);                   // ^ 반달
float sad  = fill(length(p) - .028)
           * step(p.y, .012 - p.x * side * .55); // 처진 반달
float brow = fill(abs(b.y + b.x*side*.45) - .006)
           * uScared;                           // 치켜 올라간 눈썹
q.x += sin(uTime * 47.) * .0035 * uScared;       // 두려우면 떨림
q.y /= max(uOpen, .08);                         // 깜빡임

float m = mix(base, arc, uHappy);
m = mix(m, sad, uSad);
gl_FragColor = vec4(cyan * m * 4.2, 1.);     // 1보다 밝게 → 블룸
